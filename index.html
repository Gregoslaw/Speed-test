const express = require('express'); const app = express(); const path = require('path'); // Serwujemy statyczne pliki (index.html) z głównego folderu app.use(express.static(path.join(__dirname))); // Endpoint getIP app.get('/getIP.php', (req, res) => { const ip = req.headers['x-forwarded-for'] || req.socket.remoteAddress; res.send(ip.replace('::ffff:', '')); }); // Endpoint empty (do wysyłania/pingu) app.get('/empty.php', (req, res) => { res.status(200).send(''); }); // Endpoint garbage (do pobierania) app.get('/garbage.php', (req, res) => { // Generujemy 1MB "śmieci" do testu pobierania const garbage = Buffer.alloc(1 * 1024 * 1024, '0'); res.set('Content-Type', 'application/octet-stream'); res.send(garbage); }); const listener = app.listen(process.env.PORT || 3000, () => { console.log('Test prędkości działa na porcie ' + listener.address().port); });
